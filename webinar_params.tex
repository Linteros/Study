\sectitle{1}{Пример 1}
\begin{problem}
Найдите все значения параметра $a$, при каждом из которых система
\[
\begin{cases}
\dfrac{(y^2-xy+2x-2y)\sqrt{x+1}}{\sqrt{6-x}}=0,\\[2mm]
x+y-a-1=0
\end{cases}
\]
имеет ровно два различных решения.
\end{problem}

\subhead{Решение}
\begin{rulebox}
\[
\frac{A}{B}=0\iff\begin{cases}A=0,\\B\ne0,\end{cases}
\qquad
\sqrt A\ \text{существует}\iff A\ge0,
\qquad
\frac1{\sqrt A}\ \text{существует}\iff A>0.
\]
\end{rulebox}

Рассмотрим первое уравнение системы:
\[
\begin{cases}
(y^2-xy+2x-2y)\sqrt{x+1}=0,\\
6-x>0.
\end{cases}
\]
Так как
\[
y^2-xy+2x-2y=(y-2)(y-x),
\]
получаем
\[
(y-2)(y-x)\sqrt{x+1}=0.
\]
Из ограничений
\[
6-x>0\iff x<6,
\qquad
x+1\ge0\iff x\ge-1.
\]
Следовательно,
\[
\boxed{
\begin{cases}
-1\le x<6,\\[1mm]
\left[\begin{aligned}
y&=2,\\
y&=x,\\
x&=-1.
\end{aligned}\right.
\end{cases}}
\]

Второе уравнение:
\[
x+y-a-1=0\iff \boxed{y=-x+a+1}.
\]
Это прямая с угловым коэффициентом $k=-1$, проходящая через точку $(0;a+1)$.

\sectitle{2}{Пример 2}
\begin{problem}
Найдите все значения параметра $a$, при каждом из которых система
\[
\begin{cases}
y^2-10y-x^2+6x+16=0,\\
y-ax=1,\\
x>1
\end{cases}
\]
имеет единственное решение.
\end{problem}

\subhead{Решение}
Выделим полные квадраты:
\[
y^2-10y+25=(y-5)^2,
\qquad
x^2-6x+9=(x-3)^2.
\]
Тогда первое уравнение принимает вид
\[
(y-5)^2-(x-3)^2=0,
\]
то есть
\[
(y-5-(x-3))(y-5+(x-3))=0.
\]
Отсюда
\[
\left[\begin{aligned}
y&=x+2,\\
y&=-x+8.
\end{aligned}\right.
\]
Второе уравнение:
\[
y-ax=1\iff \boxed{y=ax+1}.
\]
Следовательно, система имеет вид
\[
\begin{cases}
\left[\begin{aligned}
y&=x+2,\\
y&=-x+8,
\end{aligned}\right.\\[1mm]
y=ax+1,\\
x>1.
\end{cases}
\]

\begin{problem}
Найдите все значения параметра $a$, при каждом из которых уравнение
\[
\frac{4x^2-a^2}{x^2+6x+9-a^2}=0
\]
имеет единственное решение.
\end{problem}

\subhead{Решение}
\[
\frac{A}{B}=0\iff\begin{cases}A=0,\\B\ne0.\end{cases}
\]
Поэтому
\[
\begin{cases}
4x^2-a^2=0,\\
x^2+6x+9-a^2\ne0
\end{cases}
\iff
\begin{cases}
(2x-a)(2x+a)=0,\\
(x+3-a)(x+3+a)\ne0.
\end{cases}
\]
Перенесём в систему координат $xOa$:
\[
\boxed{
\begin{cases}
\left[\begin{aligned}
a&=2x,\\
a&=-2x,
\end{aligned}\right.\\[1mm]
a\ne x+3,\\
a\ne -x-3.
\end{cases}}
\]

A:&\quad a=-2x,\ a=-x-3
&&\Rightarrow -2x=-x-3\Rightarrow x=3,\ a=-6,\\[1mm]
B:&\quad a=2x,\ a=-x-3
&&\Rightarrow 2x=-x-3\Rightarrow x=-1,\ a=-2,\\[1mm]
C:&\quad a=2x,\ a=-2x
&&\Rightarrow x=0,\ a=0,\\[1mm]
D:&\quad a=-2x,\ a=x+3
&&\Rightarrow -2x=x+3\Rightarrow x=-1,\ a=2,\\[1mm]
E:&\quad a=2x,\ a=x+3
&&\Rightarrow 2x=x+3\Rightarrow x=3,\ a=6.
\end{aligned}
\]
